C) CONTROLLER TOKEN HELPER (DOCUMENT CONTROLLER)
================================================================
Inject AuthService into DocumentManageController:

private final AuthService authService;

public DocumentManageController(DocumentManageService documentManageService, AuthService authService) {
    this.documentManageService = documentManageService;
    this.authService = authService;
}

Add helper in controller:

private UserManager requireCurrentUser(String authorization) {
    if (authorization == null || !authorization.startsWith("Bearer ")) {
        throw new IllegalArgumentException("Unauthorized");
    }
    String token = authorization.substring(7).trim();
    return authService.getCurrentUser(token);
}

public List<DocumentManage> getDocuments(Integer currentUserId, Integer folderId, String q) {
    if (folderId != null) {
        if (q != null && !q.isBlank()) {
            return documentManageRepository
                    .findByOwner_UserIdAndFolder_FolderId(currentUserId, folderId)
                    .stream()
                    .filter(doc -> doc.getTitle() != null && doc.getTitle().toLowerCase().contains(q.toLowerCase()))
                    .toList();
        }
        return documentManageRepository.findByOwner_UserIdAndFolder_FolderId(currentUserId, folderId);
    }

    if (q == null || q.isBlank()) {
        return documentManageRepository.findByOwner_UserId(currentUserId);
    }
    return documentManageRepository.findByOwner_UserIdAndTitleContainingIgnoreCase(currentUserId, q);
}

public DocumentManage getDocumentById(Integer id, Integer currentUserId) {
    DocumentManage doc = documentManageRepository
            .findById(id)
            .orElseThrow(() -> new IllegalArgumentException("Document not found with id: " + id));

    boolean isOwner = doc.getOwner() != null && doc.getOwner().getUserId().equals(currentUserId);
    if (!isOwner) {
        throw new IllegalArgumentException("Document not found with id: " + id);
    }
    return doc;
}

public DocumentManage moveDocument(Integer id, Integer folderId, Integer currentUserId) {
    if (folderId == null) {
        throw new IllegalArgumentException("folder_id is required");
    }

    DocumentManage doc = getDocumentById(id, currentUserId);

    FolderManager target = folderManagerRepository
            .findById(folderId)
            .orElseThrow(() -> new IllegalArgumentException("Folder not found with id: " + folderId));

    boolean folderOwnedByCurrent = target.getOwner() != null && target.getOwner().getUserId().equals(currentUserId);
    if (!folderOwnedByCurrent) {
        throw new IllegalArgumentException("You can only move to your own folders");
    }

    doc.setFolder(target);
    return documentManageRepository.save(doc);
}

================================================================
G) IMPORTANT ABOUT YOUR AuthService.toUserManager
================================================================
Your AuthService.getCurrentUser(token) returns toUserManager(user), which builds a detached simplified object.
That is fine for reading userId/role/name in controllers.

If you later need fully-managed entity relations, return user directly instead of toUserManager(user).
For current owner-check flow, current.getUserId() is enough.
